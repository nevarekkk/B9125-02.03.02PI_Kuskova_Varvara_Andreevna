program OneFullIf; var x: integer; begin writeln('Введите целое число x:'); readln(x); if x > 0 then writeln('x положительное') else writeln('x не положительное (ноль или отрицательное)'); end. program TwoFullIfs; var a, b: integer; begin writeln('Введите два целых числа a и b:'); readln(a, b); if a > b then writeln('Максимум: ', a) else writeln('Максимум: ', b); if a = b then writeln('Числа равны') else writeln('Числа не равны'); end. program ThreeFullIfs; var a, b, c: integer; begin writeln('Введите три стороны a, b, c:'); readln(a, b, c); if (a + b > c) and (a + c > b) and (b + c > a) then begin writeln('Треугольник существует'); { 2-й полный if: равносторонний или нет } if (a = b) and (b = c) then writeln('Он равносторонний') else begin { 3-й полный if: равнобедренный или разносторонний } if (a = b) or (a = c) or (b = c) then writeln('Он равнобедренный') else writeln('Он разносторонний'); end; end else writeln('Треугольник не существует'); end.
